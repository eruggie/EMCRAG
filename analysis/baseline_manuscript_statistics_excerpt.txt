audit={'baseline_sha256':hashlib.sha256((DATA/'ragas_scores_per_record.csv').read_bytes()).hexdigest(),
       'baseline_n':len(f),'scatter_n':len(plot_data),'scatter_jitter':False,

ps=[]
for m in ['context_precision','context_recall','faithfulness','response_relevancy','noise_sensitivity']:
    a=f[f.dataset==styles[0][0]][m].dropna().to_numpy()
    b=f[f.dataset==styles[1][0]][m].dropna().to_numpy()
    u,p=mannwhitneyu(a,b,alternative='two-sided')
    audit['metrics'][m]={'n_IMB':len(a),'n_MMMED':len(b),'mean_IMB':float(a.mean()),
                        'mean_MMMED':float(b.mean()),'cliffs_delta':float(2*u/(len(a)*len(b))-1),
                        'mann_whitney_p':float(p)}
    ps.append((m,p))
previous=0
for i,(m,p) in enumerate(sorted(ps,key=lambda t:t[1])):
    previous=max(previous,min(1.,p*(len(ps)-i)))
    audit['metrics'][m]['holm_p']=float(previous)
for name,t in [('pooled',f)]+list(f.groupby('dataset')):
    d=t[['context_precision','faithfulness']].dropna()
    audit.setdefault('spearman_precision_faithfulness',{})[name]={'n':len(d),'rho':float(spearmanr(d.iloc[:,0],d.iloc[:,1]).statistic)}
